How to fetch live Steam API data securely without exposing your API key

Why do you need a tiny Serverless Function on Netlify or Vercel?

Valve's Steam Web API (api.steampowered.com) blocks direct browser JavaScript requests via CORS so strangers can't steal your private Steam API Key from your website's source code. By dropping the file below into your GitHub repository, Netlify (or Vercel) automatically creates a secure backend endpoint for free!

STEP 1
Push to GitHub

Save this index.html file and create netlify/functions/steam.js in your GitHub repository.

STEP 2
Import to Netlify

Log into Netlify.com or Vercel.com, click "Add New Site → Import from GitHub" and select your repo.

STEP 3
Add API Env Variable

In Netlify Site Settings → Environment Variables, add STEAM_API_KEY from steamcommunity.com/dev/apikey.

File: netlify/functions/steam.js 
// Save as: netlify/functions/steam.js (or api/steam.js on Vercel)
exports.handler = async function(event) {
  const steamId = event.queryStringParameters.steamid;
  const apiKey = process.env.STEAM_API_KEY;

  if (!steamId || !apiKey) {
    return { statusCode: 400, body: JSON.stringify({ error: "Missing steamid or STEAM_API_KEY" }) };
  }

  try {
    // 1. Fetch Player Summary (Avatar, Name, Creation Date)
    const profileUrl = `https://api.steampowered.com/ISteamUser/GetPlayerSummaries/v0002/?key=${apiKey}&steamids=${steamId}`;
    // 2. Fetch Owned Games + Playtime Forever
    const gamesUrl = `https://api.steampowered.com/IPlayerService/GetOwnedGames/v0001/?key=${apiKey}&steamid=${steamId}&include_appinfo=1&include_played_free_games=1&format=json`;

    const [profileRes, gamesRes] = await Promise.all([fetch(profileUrl), fetch(gamesUrl)]);
    const profileData = await profileRes.json();
    const gamesData = await gamesRes.json();

    return {
      statusCode: 200,
      headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" },
      body: JSON.stringify({
        player: profileData.response?.players?.[0] || null,
        games: gamesData.response?.games || []
      })
    };
  } catch (err) {
    return { statusCode: 500, body: JSON.stringify({ error: err.message }) };
  }
};
